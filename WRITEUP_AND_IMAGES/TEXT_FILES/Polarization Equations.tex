\subsection{Polarized Intensity and Error}

\begin{equation}
    POLI = \sqrt{Q^2 + U^2}
    \label{eq: POLI}
\end{equation}


\subsection{Polarized Fraction}
The linear polarization fraction,
\begin{equation}
    POLF = \sqrt{Q^2+U^2}/I,
    \label{eq: POLF}
\end{equation}
\noindent where $I$, $Q$ and $U$ are Stokes parameters
\subsection{Polarization Angle}

The poilarization position angle, $\theta$ is calculated by: \add{cite}. 
\begin{equation}
\theta = \half \tan ^{-1} \frac{U}{Q},
    \label{eq: PA}
\end{equation}
\noindent where $U$ is \SU and $Q$ is \SQ.
